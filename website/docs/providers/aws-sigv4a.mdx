## Get started

```bash
cargo add reqsign --features aws-v4a
```

```rust
use reqsign::aws::v4a::{default_signer, SigningRegionSet};

// Same credential chain as SigV4; the signature covers a region set —
// built for multi-region endpoints like S3 Multi-Region Access Points.
let region_set = SigningRegionSet::new("us-east-1,us-west-2")?;
let signer = default_signer("s3", region_set);

let mut req = http::Request::builder()
    .method(http::Method::GET)
    .uri("https://m1abcdefgh.mrap.accesspoint.s3-global.amazonaws.com/my-object")
    .body(())?
    .into_parts()
    .0;

// Sign in place, then send with the HTTP client you already use.
signer.sign(&mut req, None).await?;
```
