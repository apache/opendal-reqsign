## Get started

```bash
cargo add reqsign --features huaweicloud
```

```rust
use reqsign::huaweicloud;

// Env and config credentials — resolved by the default chain.
let signer = huaweicloud::default_signer("my-bucket");

let mut req = http::Request::builder()
    .method(http::Method::GET)
    .uri("https://my-bucket.obs.cn-north-4.myhuaweicloud.com/my-object")
    .body(())?
    .into_parts()
    .0;

// Sign in place, then send with the HTTP client you already use.
signer.sign(&mut req, None).await?;
```
